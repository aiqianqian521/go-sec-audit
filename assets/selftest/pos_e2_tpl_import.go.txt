package sample

import (
	"html/template"
	"net/http"
)

// Positive control for the file-level gate: html/template is imported and the
// argument to Parse is not a literal, so this is real template evaluation.
// Without this sample, breaking file_all_of would go undetected.
func render(w http.ResponseWriter, body string) error {
	t, err := template.New("page").Parse(body)
	if err != nil {
		return err
	}
	return t.Execute(w, nil)
}

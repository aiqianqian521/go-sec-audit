package sample

import (
	"github.com/golang-jwt/jwt/v5"
	"github.com/robfig/cron/v3"
)

// A Parse call is not template evaluation. jwt.Parse, cron.Parse and hand-rolled
// parsers are all named Parse, yet they never evaluate Go code.
// File-level criterion: without an import of text/template or html/template this
// is not E2.
// woodpecker shared/token/token.go:52 and server/cron/cron.go:81 are these two shapes.
func verify(raw string, keyFunc jwt.Keyfunc) (*jwt.Token, error) {
	return jwt.Parse(raw, keyFunc)
}

func next(schedule string) error {
	parser, err := cron.NewDefaultParser(cron.StandardOptions)
	if err != nil {
		return err
	}
	_, err = parser.Parse(schedule)
	return err
}

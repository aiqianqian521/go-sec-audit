// Negative sample: every FuncMap entry is a pure function, and the same file also
// holds a constant exec call.
// E2-FUNCMAP-DEF must stay silent (neither the FuncMap keys nor the values carry
// execution capability), and the constant exec call must not be hit by E1 either
// (the none_of constant filter).
var safeFmap = template.FuncMap{
	"upper": func(s string) string { return strings.ToUpper(s) },
}

}

package selftest

// Positive sample: shell interpreter plus a concatenated script string. Rule
// E1-SHELL-C must fire.
import (
	"net/http"
	"os/exec"
)

func ReportHandler(w http.ResponseWriter, r *http.Request) {
	name := r.FormValue("filename")
	cmd := exec.Command("sh", "-c", "cat /reports/"+name)
	out, _ := cmd.Output()
	w.Write(out)
}

package selftest

// Negative sample: arguments passed separately, no shell involved. Every E1-class
// rule must stay silent.
// Note: this is not "absolutely safe" -- only E1 (shell concatenation) does not
// hold here; argument injection is judged separately.
import (
	"net/http"
	"os/exec"
)

func SafeCat(w http.ResponseWriter, r *http.Request) {
	name := r.FormValue("filename")
	out, _ := exec.Command("cat", "/reports/"+name).Output()
	w.Write(out)
}

package selftest

// Negative sample: shell plus -c, but the script string is a compile-time
// literal, so no variable ever enters it and injection is impossible.
// Every E1-class rule must stay silent.
// Origin: on a 760k-LOC real corpus, the idiom that reads the current user's home
// directory burned a critical for nothing. Exclusion criterion: the third argument
// is a complete literal and the paren closes immediately after it.
// Note: this comment deliberately does not spell out that call shape, otherwise
// the scanner would read the comment text as a hit.
import (
	"os/exec"
)

func HomeDir() string {
	out, _ := exec.Command("sh", "-c", "eval echo ~$USER").Output()
	return string(out)
}

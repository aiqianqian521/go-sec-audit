package sample

import "os/exec"

// In `git -c key=value` the -c is a config flag, not the shell's script switch.
// The criterion is what follows "-c": a literal is a config value, a non-literal
// is a script handed to the shell.
func status() ([]byte, error) {
	return exec.Command("git", "-c", "core.hooksPath=/dev/null", "status").Output()
}

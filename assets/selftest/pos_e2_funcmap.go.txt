package selftest

// Positive sample: the FuncMap registers an executable function and the template
// text comes from the request. E2-FUNCMAP-EXEC must fire.
import (
	"net/http"
	"os/exec"
	"text/template"
)

func RenderUserTpl(w http.ResponseWriter, r *http.Request) {
	tpl := r.URL.Query().Get("tpl")
	t := template.New("user")
	t.Funcs(template.FuncMap{
		"run": func(c string) string {
			out, _ := exec.Command("sh", "-c", c).CombinedOutput()
			return string(out)
		},
	})
	t.Parse(tpl)
	t.Execute(w, nil)
}

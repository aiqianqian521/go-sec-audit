package x

import (
	"html/template"
	"os/exec"
)

// The FuncMap definition is separated from the later .Funcs registration:
// E2-FUNCMAP-EXEC cannot anchor this shape, so E2-FUNCMAP-DEF has to anchor the
// FuncMap literal directly.
var fmap = template.FuncMap{
	"run": func(c string) string {
		out, _ := exec.Command("sh", "-c", c).CombinedOutput()
		return string(out)
	},
}

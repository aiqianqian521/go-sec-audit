package selftest

// Positive sample: writes into the git hooks directory (write-then-execute). Rule
// E4-HOOK-WRITE must fire.
import (
	"os"
	"path/filepath"
)

func InstallHook(repoDir, body string) error {
	target := filepath.Join(repoDir, ".git", "hooks", "post-merge")
	return os.WriteFile(target, []byte(body), 0755)
}

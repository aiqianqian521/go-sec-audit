package sample

import (
	"os"
	"os/exec"
)

// The shell is not a literal but lives in a parameter/variable: E1-SHELL-C demands
// a literal, so this shape falls outside it.
// woodpecker pipeline/backend/local/command.go:182 is exactly this shape.
func runStep(shell, script string) error {
	cmd := exec.Command(shell, "-e", "-c", script)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	return cmd.Run()
}

// Forwarded by a wrapper: the literal shell appears in a call that is not
// exec.Command.
// woodpecker pipeline/backend/local/clone.go:103 is exactly this shape.
func runWrapped(ctx interface{}, pluginGit, rmCmd string) {
	_ = newCmd(ctx, "/bin/sh", "-c", pluginGit+"; "+rmCmd)
}

func newCmd(ctx interface{}, name string, args ...string) error {
	return nil
}

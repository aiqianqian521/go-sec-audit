package driver

import "context"

// The interface method happens to be named Extract, but it is not archive
// extraction and carries no Zip Slip semantics.
// A blind test found alist's 22 Tier B hits and cloudreve's 18 Tier B hits were all
// of this kind: a driver.Extract method definition misread as an extraction
// vulnerability. Hence this negative sample.
type Extractor interface {
	Extract(ctx context.Context, obj []byte, args Args) ([]byte, error)
}

type Args struct{}
type Gofile struct{}

func (d *Gofile) Extract(ctx context.Context, obj []byte, args Args) ([]byte, error) {
	return obj, nil
}

type Alias struct{}

func (d *Alias) Extract(ctx context.Context, obj []byte, args Args) ([]byte, error) {
	return obj, nil
}

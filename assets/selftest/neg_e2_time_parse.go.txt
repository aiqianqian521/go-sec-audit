package ingest

import (
	"net/url"
	"time"
)

// Parsing a timestamp is not template evaluation: time.Parse only returns a
// time.Time and can never evaluate code.
// A blind test found 18 of gogs' 21 Tier A hits were exactly this false positive,
// which is why this negative sample exists.
func Since(raw string) (time.Time, error) {
	return time.Parse(time.RFC3339, raw)
}

// Parsing a URL is not template evaluation either: url.Parse only returns *url.URL.
func Target(raw string) (*url.URL, error) {
	u, err := url.Parse(raw)
	if err != nil {
		return nil, err
	}
	return u, nil
}

// Same for parsing a host address: the product is an IP, not executable text.
func Host(raw string) string {
	u, err := url.Parse(raw)
	if err != nil {
		return ""
	}
	return u.Host
}

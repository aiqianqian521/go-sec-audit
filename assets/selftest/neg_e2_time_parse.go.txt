package ingest

import (
	"net/url"
	"time"
)

// 时间解析不是模板求值：time.Parse 只返回 time.Time，永远评不出代码。
// 盲测发现 gogs 的 21 条 Tier A 里有 18 条是这类误报，故立此负样本。
func Since(raw string) (time.Time, error) {
	return time.Parse(time.RFC3339, raw)
}

// 解析 URL 不是模板求值：url.Parse 只返回 *url.URL。
func Target(raw string) (*url.URL, error) {
	u, err := url.Parse(raw)
	if err != nil {
		return nil, err
	}
	return u, nil
}

// 解析主机地址同理，产出的是 IP 不是可执行文本。
func Host(raw string) string {
	u, err := url.Parse(raw)
	if err != nil {
		return ""
	}
	return u.Host
}
